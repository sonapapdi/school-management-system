    61→/** Allocate amount across unpaid charges, oldest due first. */
    62→export function allocateCharges(
    63→  open: { id: string; amount: number; dueDate: string }[],
    64→  amount: number,
    65→): { chargeId: string; amount: number }[] {
    66→  const sorted = [...open].sort((a, b) => a.dueDate.localeCompare(b.dueDate));
    67→  let left = amount;
    68→  const out: { chargeId: string; amount: number }[] = [];
    69→  for (const c of sorted) {
    70→    if (left <= 0) break;
    71→    const pay = Math.min(left, c.amount);
    72→    if (pay > 0) out.push({ chargeId: c.id, amount: pay });
    73→    left -= pay;
    74→  }
    75→  return out;
    76→}
