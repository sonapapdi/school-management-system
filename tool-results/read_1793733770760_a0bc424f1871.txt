   125→  getItem: (name) => {
   126→    try {
   127→      return localStorage.getItem(name);
   128→    } catch {
   129→      return null;
   130→    }
   131→  },

   133→    try {

   135→    } catch {
   136→      /* quota exceeded — prototype continues in memory */
   137→    }
   138→  },
   139→  removeItem: (name) => {
   140→    try {
